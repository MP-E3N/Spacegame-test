```js
// form.js (PL) — Form bekommt bauteile; normalisiere kennt den Katalog NICHT (keine Zirkel-Imports), nur die Form:
//   form.bauteile = [{ id (Ganzzahl), typ (Katalog-Schlüssel), p: [x, y, z] (Fußpunkt AUF der Fläche),
//                      n: [Achse ±1] (Flächennormale nach außen), spiegel: [] | ['x'] … (wie Körper), fest? }]
// geschuetze.js (Waffentechniker):
export const GESCHUETZE = { pdc: {…}, flak: {…}, schwer: {…} }
//   je { art: 'geschuetz', name, masse (t), turm: { typ: 'flak' | 'schwer', s }, gier: [mitte, halbweite], nick: [min, max],
//        flaeche: ['oben', 'unten'] }
export function turmAusBauteil(b, id)   // → Turm-Datensatz wie schiffe/turm.js turmDaten (x, y, z, oben, typ, s, sockel,
                                        //   schuerze, zapfen, muendungen, gier, nick) — nur bei n = ±Y, sonst null
export function pruefeGeschuetze(form)  // → [{ code, stufe, text, bauteil }] (Seitenmontage, im Rumpf, zu dicht …)
// flug.js (Flugsystem):
export const ANTRIEBE = { triebwerk_s: {…}, triebwerk_m: {…}, triebwerk_l: {…}, steuer: {…} }
//   je { art: 'antrieb', name, masse (t), schub (kN, entlang −n … Schub wirkt nach +Z), r (Düsenradius m), flaeche: ['hinten'] | ['alle'] }
export function flugwerte(form, katalog) // → { werte (genau die Form von WERTE.bollwerk), masse (t), schwerpunkt [x,y,z],
                                         //   schubpunkt [x,y,z] | null, schub (kN), meldungen: [{ code, stufe, text, bauteil? , koerper? }] }
// bauteile.js (PL, nach dem Merge): BAUTEILE = { ...GESCHUETZE, ...ANTRIEBE }; bauteilListe(form) (inkl. Spiegelkopien,
//   Lage gespiegelt); formZuSchiff(form) → { name, kennung, laenge, kugeln, tuerme, duesen: [{ x, y, z, r }], werte, form }
// grafik/form/schiff.js (Schiffbauer):
export function erstelleFormSchiff(scene, schiff) // schiff = formZuSchiff(form) → { name, kennung, laenge, gruppe, tuerme,
                                                  //   darstellen(pos, rot, z, zeit, dt) } — wie erstelleBollwerk
export function bauteilGrafik(b, mats)            // → THREE.Group eines Bauteils (Werkstatt-Vorschau + Schiff)
```
Koordinaten wie das Schiff: +Z Bug, +X Backbord, +Y oben; Triebwerke zeigen mit der Düse nach −Z (n = [0, 0, −1]).
